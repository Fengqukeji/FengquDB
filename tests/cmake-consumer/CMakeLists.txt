cmake_minimum_required(VERSION 3.24)
project(FengquDBConsumer LANGUAGES CXX)
find_package(FengquDB 0.3.0 CONFIG REQUIRED)
add_executable(cpp_consumer ../../examples/basic.cpp)
target_link_libraries(cpp_consumer PRIVATE FengquDB::Cpp)
add_executable(store_consumer store_consumer.cpp)
target_link_libraries(store_consumer PRIVATE FengquDB::Cpp)
if(MSVC AND NOT FengquDB_SHARED)
  set_property(TARGET cpp_consumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET store_consumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(TARGET FengquDB::Diagnostics)
  add_executable(diagnostics_consumer diagnostics_consumer.cpp)
  target_link_libraries(diagnostics_consumer PRIVATE FengquDB::Cpp FengquDB::Diagnostics)
  if(MSVC AND NOT FengquDB_SHARED)
    set_property(TARGET diagnostics_consumer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
