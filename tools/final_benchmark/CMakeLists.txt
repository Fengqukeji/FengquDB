cmake_minimum_required(VERSION 3.24)
project(FengquFinalBenchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FQDB_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(FQDB_BUILD_JNI OFF CACHE BOOL "" FORCE)
set(FQDB_BUILD_DIAGNOSTICS OFF CACHE BOOL "" FORCE)
set(FQDB_SHARED ON CACHE BOOL "" FORCE)
if(NOT FQDB_SOURCE)
  get_filename_component(FQDB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_subdirectory("${FQDB_SOURCE}" core)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MYSQL REQUIRED IMPORTED_TARGET mysqlclient)
pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis)
find_library(SQLITE_BENCH sqlite3_benchmark PATHS /opt/bench/lib REQUIRED NO_DEFAULT_PATH)
add_executable(fqdb_final_bench benchmark.cpp fq.cpp sqlite.cpp mysql.cpp redis.cpp)
target_include_directories(fqdb_final_bench PRIVATE /opt/bench/include)
target_link_libraries(fqdb_final_bench PRIVATE FengquDB::Core ${SQLITE_BENCH} PkgConfig::MYSQL PkgConfig::HIREDIS Threads::Threads)
target_compile_options(fqdb_final_bench PRIVATE -Wall -Wextra -Wpedantic)
include(CheckIPOSupported)
check_ipo_supported(RESULT BENCH_IPO LANGUAGES CXX)
if(BENCH_IPO)
  set_property(TARGET fqdb_final_bench PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
set_target_properties(fqdb_final_bench PROPERTIES BUILD_RPATH "/opt/bench/lib" INSTALL_RPATH "$ORIGIN/../lib")
